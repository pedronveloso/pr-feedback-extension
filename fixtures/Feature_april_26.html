      private fun translatableStringNames(file: Path): List<String> {
      val document = parseXml(file)
      val strings = document.getElementsByTagName("string")
      return buildList {
      for (index in 0 until strings.length) {
      val element = strings.item(index) as Element

      private fun localeNames(file: Path): List<String> {
      val document = parseXml(file)
      val locales = document.getElementsByTagName("locale")
      return buildList {
      for (index in 0 until locales.length) {
      val element = locales.item(index) as Element
      add(element.getAttribute("android:name"))
      }
      }
      }

      path() is called with paths that already start with app/…, but the fallback branch prepends another app/ ("app/$relativePath"). If the test working directory is the app module (common for Gradle), both direct and nestedApp will miss and the test will fail. Consider passing module-relative paths (e.g. src/main/res/...) and only prefixing with app/ in the fallback, or make path() normalize/strip a leading app/ before trying the nested lookup.
